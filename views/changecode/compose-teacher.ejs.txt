        </div>
        <input type="hidden" name="audienceType" id="audienceType" value="select" />

        <div id="selectFields">
          <div class="msg-field">
            <label for="teacherFilterBox">Filter list</label>
            <input type="text" id="teacherFilterBox" placeholder="Type to filter by name or mobile…" />
          </div>

          <div class="msg-field" style="max-height: 260px; overflow-y: auto; border: 1px solid var(--slate-300); border-radius: var(--radius); padding: 8px;">
            <% if (!teachers || teachers.length === 0) { %>

                <input type="checkbox" name="teacherIds" value="<%= t._id %>" />

<script>
(function () {
  const segButtons = document.querySelectorAll(".msg-segmented button");
  const audienceInput = document.getElementById("audienceType");
  const selectFields = document.getElementById("selectFields");

  segButtons.forEach((btn) => {
    btn.addEventListener("click", () => {
      segButtons.forEach((b) => b.classList.remove("active"));
      btn.classList.add("active");
      audienceInput.value = btn.dataset.audience;
      selectFields.classList.toggle("msg-hidden", btn.dataset.audience === "all");
    });
  });

  const filterBox = document.getElementById("teacherFilterBox");
  filterBox.addEventListener("input", () => {
    const q = filterBox.value.trim().toLowerCase();
    document.querySelectorAll(".teacherRow").forEach((row) => {
      row.style.display = row.dataset.search.includes(q) ? "flex" : "none";
    });
  });

  const checkboxes = document.querySelectorAll('input[name="teacherIds"]');
  const selectedCount = document.getElementById("selectedCount");
  function updateCount() {
    const n = Array.from(checkboxes).filter((c) => c.checked).length;
    selectedCount.textContent = `${n} selected`;
  }
  checkboxes.forEach((c) => c.addEventListener("change", updateCount));

  document.getElementById("composeForm").addEventListener("submit", (e) => {
    if (audienceInput.value === "select") {
      const anyChecked = Array.from(checkboxes).some((c) => c.checked);
      if (!anyChecked) {
        e.preventDefault();
        alert("Please select at least one teacher.");
      }
    }
  });
})();
</script>
